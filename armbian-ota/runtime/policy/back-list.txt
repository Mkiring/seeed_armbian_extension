# OTA local configuration preserve whitelist (absolute paths)
# Lines starting with # are comments.

# Keep the preserve list itself across OTA.
/etc/armbian-ota/back-list.txt

# Keep OTA persist data (account files + /home) across recovery OTA. In recovery
# mode /userdata is a rootfs directory, so it would be wiped by the rootfs
# rewrite; preserving it lets the /userdata/.persist bind mounts carry user
# accounts and home across the update. (A/B mode keeps .persist on the separate
# armbi_usrdata partition, which is never rewritten, so this entry is a no-op there.)
/userdata/.persist

/etc/default/locale

/etc/default/keyboard
/etc/environment

# Keep stable device identity and local name settings.
/etc/hostname
/etc/hosts
/etc/machine-id

# Keep SSH host identity so clients do not see a host key change after OTA.
# Wildcards are intentionally not used because the preserve helper rejects them.
/etc/ssh
